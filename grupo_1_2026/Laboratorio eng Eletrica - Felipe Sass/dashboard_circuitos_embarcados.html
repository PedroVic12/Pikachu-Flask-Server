💻 Código C++ Arduino: Controle dos LEDs por Luminosidade (LDR)

// ===================================================================
// LUMINOSIDADE DO QUARTO (LDR) E SELEÇÃO DE LEDS: VERMELHO, AMARELO, VERDE
// ===================================================================

#define PIN_LDR A1       // Pino do sensor de luz (LDR)
#define PIN_TEMP A2      // Pino do sensor de temperatura

#define LED_RED 9        // LED Vermelho -> Quarto Escuro (< 300 ADC)
#define LED_YELLOW 10    // LED Amarelo -> Meia Luz (300 a 700 ADC)
#define LED_GREEN 11     // LED Verde -> Quarto Claro (> 700 ADC)

void setup() {
    Serial.begin(9600);
    pinMode(LED_RED, OUTPUT);
    pinMode(LED_YELLOW, OUTPUT);
    pinMode(LED_GREEN, OUTPUT);
}

void loop() {
    int adcLdr = analogRead(PIN_LDR);
    
    // Lógica de Seleção dos LEDs baseada na Luminosidade
    if (adcLdr < 300) {
        // QUARTO ESCURO: Acende apenas o LED Vermelho
        digitalWrite(LED_RED, HIGH);
        digitalWrite(LED_YELLOW, LOW);
        digitalWrite(LED_GREEN, LOW);
    } else if (adcLdr >= 300 && adcLdr <= 700) {
        // MEIA LUZ: Acende apenas o LED Amarelo
        digitalWrite(LED_RED, LOW);
        digitalWrite(LED_YELLOW, HIGH);
        digitalWrite(LED_GREEN, LOW);
    } else {
        // QUARTO CLARO: Acende apenas o LED Verde
        digitalWrite(LED_RED, LOW);
        digitalWrite(LED_YELLOW, LOW);
        digitalWrite(LED_GREEN, HIGH);
    }

    // Transmissão das Séries Temporais via Porta Serial (CSV)
    Serial.print("LDR:"); Serial.println(adcLdr);
    delay(500);
}
