IMPORTANTE: Mencionem a função do BMS. VL53L0X Sensor de Distância Laser.

1. Arquitetura geral

                 PACK 2S Li-Ion
                   7,4 V nominal
                       │
                 ┌─────┴─────┐
                 │           │
                 ▼           ▼
             L298N        Buck 5 V
           alimentação      │
             motores        │
                 │          ▼
              Motores     Arduino Nano
                             │
                ┌─────────────┼─────────────┐
                │             │             │
                ▼             ▼             ▼
             HC-SR04        QRE          KY-022
           ultrassônico    reflexão       IR

Todos os GND precisam estar em comum.

2. Arduino Nano → L298N

Arduino Nano L298N Função
D5 ENA PWM Motor A
D7 IN1 Direção A
D8 IN2 Direção A
D6 ENB PWM Motor B
D9 IN3 Direção B
D10 IN4 Direção B
GND GND Ground

Motores

L298N Motor
OUT1 Motor A fio 1
OUT2 Motor A fio 2
OUT3 Motor B fio 1
OUT4 Motor B fio 2

Se um motor estiver girando invertido, inverta a polaridade dos fios daquele motor.

3. Alimentação do Arduino Nano

Bateria 7,4 V ──► Buck 5 V ──┬──► Nano 5V
                             └──► Sensores 5V

Entrada e Saída:

Buck 5V  ─────► Nano 5V
Buck GND ─────► Nano GND

E o GND continua comum ao aterramento:

Nano GND ──┬── L298N GND
           ├── HC-SR04 GND
           ├── QRE GND
           └── KY-022 GND

Os sinais PWM e sensores têm a mesma referência elétrica.

4. HC-SR04 - Sensor Ultrassônico

Usando:

HC-SR04 Arduino Nano
VCC 5V
GND GND
TRIG D2
ECHO D3

Ligações:

VCC  ─────► 5V
GND  ─────► GND
TRIG ─────► D2
ECHO ─────► D3

6. Sensor de refletância QRE

Como você especificou uma versão analógica, podemos usar A0.

QRE Arduino Nano
VCC 5V
GND GND
AO A0

Ligações:

VCC ───► 5V
GND ───► GND
AO  ───► A0

No Arduino:

int leitura = analogRead(A0);

7. Receptor IR KY-022 / VS1838B

KY-022 Arduino
S D4
+ 5V
- GND

- [x] 3 Sensor Ultrassônico

Façam um código que envie pelo terminal serial a distância lida pelo sensor ultrassônico a cada 1s. Mostrem para um professor que conseguiram cumprir a missão e expliquem o funcionamento do código.

// Variáveis
#define trig 2
#define echo 3

void setup() {
  Serial.begin(9600);
  // Configurando os pinos digitais
  pinMode(trig, OUTPUT);
  pinMode(echo, INPUT);
}

void loop() {
  long duracao;
  float distancia;

  // Gerando o pulso da Onda Digital do sensor ultrassônico
  digitalWrite(trig, LOW);
  delayMicroseconds(2);
  digitalWrite(trig, HIGH);
  delayMicroseconds(10);
  digitalWrite(trig, LOW);

  // Medindo o tempo de retorno
  duracao = pulseIn(echo, HIGH);

  // Calcula a distância em cm
  float vel_som = 0.0343;
  distancia = duracao * vel_som / 2;

  Serial.print("Distancia: ");
  Serial.print(distancia);
  Serial.println(" cm");
  delay(1000);
}

- [ ] 4 Driver e Motores

Façam um código que controle o funcionamento dos motores de acordo com o fluxograma anexado nesta atividade. Mostrem para um professor que conseguiram cumprir a missão e expliquem o funcionamento do código.

/*
* Arduino Nano + L298N
* 2 motores DC
*/

// Variáveis e Configurações
const int PWM_max = 255;
const int PWM_50 = 128;
const unsigned long TEMPO_ESPERA = 5000; // 5 segundos

// Motor esquerdo
const int PWM_MOTOR_ESQUERDO = 5;
const int IN1_MOTOR_ESQUERDO = 7;
const int IN2_MOTOR_ESQUERDO = 8;

// Motor direito
const int PWM_MOTOR_DIREITO = 6;
const int IN3_MOTOR_DIREITO = 9;
const int IN4_MOTOR_DIREITO = 10;

// ======================================================
// FUNÇÕES DOS MOTORES
// ======================================================
void moverFrente(int velocidadePWM) {
  /*
  * MOTOR ESQUERDO
  * Isso cria uma polaridade no motor, considerando a ligação dos fios,
  * faz a roda esquerda girar para frente.
  */
  digitalWrite(IN1_MOTOR_ESQUERDO, HIGH);
  digitalWrite(IN2_MOTOR_ESQUERDO, LOW);

  /*
  * MOTOR DIREITO
  */
  digitalWrite(IN3_MOTOR_DIREITO, HIGH);
  digitalWrite(IN4_MOTOR_DIREITO, LOW);

  /*
  * ENA e ENB recebem PWM.
  * 0 = motor desligado
  * 128 ≈ 50%
  * 255 = potência máxima
  */
  analogWrite(PWM_MOTOR_ESQUERDO, velocidadePWM);
  analogWrite(PWM_MOTOR_DIREITO, velocidadePWM);
}

void moverRe(int velocidadePWM) {
  /*
  * MOTOR ESQUERDO
  * Invertemos a polaridade:
  * IN1 = LOW
  * IN2 = HIGH
  * Agora a corrente atravessa o motor no sentido oposto ao movimento para frente.
  */
  digitalWrite(IN1_MOTOR_ESQUERDO, LOW);
  digitalWrite(IN2_MOTOR_ESQUERDO, HIGH);

  /*
  * MOTOR DIREITO
  */
  digitalWrite(IN3_MOTOR_DIREITO, LOW);
  digitalWrite(IN4_MOTOR_DIREITO, HIGH);

  // PWM
  analogWrite(PWM_MOTOR_ESQUERDO, velocidadePWM);
  analogWrite(PWM_MOTOR_DIREITO, velocidadePWM);
}

void pararMotores() {
  // PWM = 0. Retiramos a energia efetiva enviada aos motores.
  analogWrite(PWM_MOTOR_ESQUERDO, 0);
  analogWrite(PWM_MOTOR_DIREITO, 0);

  // Não existe diferença de potencial aplicada entre os terminais do motor.
  // Motor A → parado
  digitalWrite(IN1_MOTOR_ESQUERDO, LOW);
  digitalWrite(IN2_MOTOR_ESQUERDO, LOW);

  // Motor B → parado
  digitalWrite(IN3_MOTOR_DIREITO, LOW);
  digitalWrite(IN4_MOTOR_DIREITO, LOW);
}

void girarSentidoHorario(int velocidadePWM) {
  /*
  * Para girar sobre o próprio eixo,
  * fazemos uma roda andar para frente
  * enquanto a outra anda para trás.
  *
  * MOTOR ESQUERDO → FRENTE
  * IN1 = HIGH
  * IN2 = LOW
  */
  digitalWrite(IN1_MOTOR_ESQUERDO, HIGH);
  digitalWrite(IN2_MOTOR_ESQUERDO, LOW);

  /*
  * MOTOR DIREITO → RÉ
  * IN3 = LOW
  * IN4 = HIGH
  */
  digitalWrite(IN3_MOTOR_DIREITO, LOW);
  digitalWrite(IN4_MOTOR_DIREITO, HIGH);

  /*
  * PWM de ambos os motores.
  * velocidadePWM = 128 (aproximadamente 50% do duty cycle).
  */
  analogWrite(PWM_MOTOR_ESQUERDO, velocidadePWM);
  analogWrite(PWM_MOTOR_DIREITO, velocidadePWM);
}

void setup() {
  // Motor Esquerdo
  pinMode(PWM_MOTOR_ESQUERDO, OUTPUT);
  pinMode(IN1_MOTOR_ESQUERDO, OUTPUT);
  pinMode(IN2_MOTOR_ESQUERDO, OUTPUT);

  // Motor Direito
  pinMode(PWM_MOTOR_DIREITO, OUTPUT);
  pinMode(IN3_MOTOR_DIREITO, OUTPUT);
  pinMode(IN4_MOTOR_DIREITO, OUTPUT);

  // Estado inicial -> parado
  pararMotores();
  delay(1000);
}

void loop() {
  // 1. ANDAR PARA FRENTE - POTÊNCIA MÁXIMA
  moverFrente(PWM_max);
  delay(TEMPO_ESPERA);

  // 2. Parar e esperar 5 segundos
  pararMotores();
  delay(TEMPO_ESPERA);

  // 3. RÉ - POTÊNCIA MÁXIMA
  moverRe(PWM_max);
  delay(TEMPO_ESPERA);

  // 4. Parar e esperar 5 segundos
  pararMotores();
  delay(TEMPO_ESPERA);

  // 5. GIRAR NO SENTIDO HORÁRIO (PWM = 50%)
  girarSentidoHorario(PWM_50);
  delay(TEMPO_ESPERA);
}
